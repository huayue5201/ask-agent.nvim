1. DESIGN                                                   *ask-agent-design*

Agents that run outside Neovim only see what is on disk. This plugin sends the
question `and` the thing you are looking at: cwd, tab, `file:line`, visual
selection, unsaved state. Delivery is a socket or a command line, so it works
with any agent that can receive text.

2. SETUP                                                     *ask-agent-setup*

>lua
require("ask_agent").setup({
  matching = "exact",            -- "exact" | "related" | "single"
  picker = "map",                -- "map" | "list"
  list_impl = "auto",            -- "auto" | "ui_menu" | "vim_ui"
  layout = nil,                  -- layout adapter id (default: first usable one)
  keys = {},                     -- dashboard key overrides, see *ask-agent-dashboard*
  show_stale = true,             -- list exited instances too, marked "not running"
  spinner = { "⠋", "⠙", "⠹" },      -- frames for the busy icon (default: braille)
  tick = 90,                     -- animation interval in ms (default 90)
  state_interval = 2000,         -- state polling interval in ms (default 2000)
  state_notify = true,           -- notify on finished/blocked agents (default true)
  notify_style = true,           -- bold + level colour for notifications (default true)
  notify_sound = false,          -- make a sound on every notification (default false)
  notify_sound_cmd = { "afplay", "/System/Library/Sounds/Glass.aiff" },
                                 -- play this instead of the terminal bell
  notify_belloff = false,        -- true = `belloff=all` also mutes the plugin
  notify_popup = false,          -- corner card instead of one cmdline line
  notify_popup_ms = 3000,        -- how long that card stays up
  notify = {                     -- per-state copy/level/bell; false = silent
    done = { text = "[{backend}] {name} finished{msg}", sound = true },
    blocked = { text = "[{backend}] {name} is waiting for you{msg}" },
    permission = { text = "[{backend}] {name} needs your permission{msg}" },
    question = { text = "[{backend}] {name} is waiting for your answer{msg}" },
    auth = { text = "[{backend}] {name} is waiting for credentials{msg}" },
    error = { text = "[{backend}] {name} failed{msg}", level = "ERROR" },
    gone = { text = "[{backend}] {name} exited while {was}" },
  },
  notify_gap = 30000,            -- ms between done/error/gone per agent
  tree_lines = true,             -- │ ├ └ between agents and tasks (default true)
  running_line = "dash",         -- green path to the running task:
                                 -- "dash" | "pulse" | "static" | "off"
  task_limit = 5,                -- rows per agent before "… N older tasks"
                                 -- (false shows every stored task)
  task_ttl = 24 * 3600 * 1000,   -- prune finished tasks older than this (ms);
                                 -- the newest task per agent always stays
  persist = true,                -- keep tasks/unread marks in the journal (default true)
  snapshots = {                  -- turn trees for change review; false disables
    enabled = true,
    keep = 50,                   -- newest turns retained
    ttl = 7 * 24 * 3600 * 1000,  -- max age in ms
    trigger_state = true,        -- also bracket working -> done turns
  },
  review = {                     -- how |:AskAgentDiff| renders
    backend = "auto",            -- auto | diffbandit | difftool | codediff | builtin
    include_unsaved = true,      -- fold modified buffers into the end side
  },
  view = {                       -- <CR> viewer float (see |ask-agent-viewer|)
    readonly = false,            -- refuse input even after `i` (tmux -r)
    width = 0.85,                -- share of the editor a big agent may take
    height = 0.75,               -- (1 = fill the screen)
  },
  map = {                        -- map canvas and labels (|ask-agent-pickers|)
    width = 0.25,                -- share of the editor width
    min_width = 12,              -- clamp in cells
    max_width = 32,
    height = 0.20,               -- share of the editor height
    min_height = 3,
    max_height = 9,
    name = true,                 -- selected cell shows the agent name
    name_min = 3,                -- fewest columns a shortened name may take
    name_over_badge = true,      -- drop the badge before cutting the name further
  },
  history = {                    -- delivered questions; false disables
    enabled = true,
    keep = 200,                  -- newest records retained
    max_text = 20000,            -- cap per record (long pastes are truncated)
  },
  results = false,               -- also keep the agent's final answer (`o`),
                                 -- opt-in: the ledger is questions otherwise
  prompts = {                    -- prompt templates
    global_path = vim.fn.stdpath("cache") .. "/ask-agent/prompts.json",
    project_file = ".ask-agent/prompts.json",  -- relative to the project root
  },
  context_lines = 20,            -- lines of source inlined for a dirty buffer
  max_selection = 200,           -- most lines a visual selection carries
  max_selection_bytes = 8192,    -- and its byte cap
  project = {                    -- <project-overview> digest (see *ask-agent-anchor*)
    enabled = true,
    wait_ms = 1500,              -- how long the overview may take
    full = false,                -- full or outline mode
    ttl = 300,                   -- seconds a digest is reused
  },
  default = "pi",                -- default backend id
  extra = {},                    -- backends registered at setup time
  icons = {                      -- every glyph: flat key = an agent icon,
    pi = "π",                    -- "ui" group = the interface (see *ask-agent-icons*)
    ui = { state = { done = "✓" }, spinner = { "⠋", "⠙" } },
  },
  icon_colors = {},              -- per-agent brand colors (hex)
  aliases = { file = vim.fn.stdpath("cache") .. "/ask-agent/aliases.json" },
  bridge = { dir = vim.fn.stdpath("cache") .. "/pi-ctx" },  -- false disables
})
<

                                                 *ask-agent-icons*
Icons and marks
Everything the plugin draws is configured through the `icons` option and defined
in `lua/ask_agent/icons.lua`:

  flat key       an agent icon, keyed by backend or instance id; `fallback` is
                 used for unknown agents
  ui.header      dashboard winbar and notification prefix
  ui.state       blocked done error working idle unknown gone
  ui.task        queued done error (dashboard tree)
  ui.tree        collapsed expanded leaf, plus the connectors vertical branch
                 corner horizontal
  ui.spinner     busy frames; the list replaces the default frames entirely

  >lua
  require("ask_agent").setup({
    icons = { claude = "", ui = { state = { blocked = "!" } } },
    icon_colors = { claude = "#d97757" },
  })
  <
Only the keys you pass change; everything else keeps its default. A top-level
`spinner` option still wins over `ui.spinner`.

3. COMMANDS                                               *ask-agent-commands*

                                                        *:AskAgent*
:AskAgent [text]        Ask the current backend; the cursor position is the
                        anchor. Without [text] the ask window opens.

                                                  *:AskAgentVisual*
:AskAgentVisual         Ask with the visual selection as anchor.

                                                  *:AskAgentPi*
:AskAgentPi             Ask the `pi` backend explicitly.

                                                  *:AskAgentBackend*
:AskAgentBackend [id]   Switch the default backend, or pick one.

                                                  *:AskAgentPicker*
:AskAgentPicker [mode]  Choose how multiple candidates are presented.

                                                  *:AskAgentTarget*
:AskAgentTarget [id]    Pin the target instance (skip matching); `!` clears.

                                                  *:AskAgentRename*
:AskAgentRename         Name an instance; names show up in lists, prompts and
                        notifications. `!` clears a name.

                                                  *:AskAgentDashboard*
:AskAgentDashboard      Open the sidebar: agents as a tree, each agent's queued
                        and sent tasks below it. `<CR>` views the agent in a float (or jumps
                        to the anchor of a task another tool dispatched),
                        `J` jumps out to the agent's own window
                        (rex/tmux), `s` send, `S` flush its queued questions,
                        `a` acknowledge, `d` delete a queued task or dismiss a
                        dead agent, `<Tab>` fold,
                        `?` key help, `r` poll now, `q` close.

                                                  *:AskAgentBatch*
:AskAgentBatch          Queue a question at the cursor instead of sending it.

                                                  *:AskAgentBatchVisual*
:AskAgentBatchVisual    Queue a question anchored to the visual selection.

                                                  *:AskAgentBatchSend*
:AskAgentBatchSend [ids]
                        Send queued questions, one message each. Routed entries
                        go to their own agent; the unrouted rest shares one
                        target picker. `ids` picks a subset (`3`, `3,5-7`);
                        without it every entry is sent. Successful entries leave
                        the queue; failed ones stay for a retry.

                                                  *:AskAgentBatchRoute*
:AskAgentBatchRoute [ids]
                        Route queued questions to one target (`ids` as above;
                        without it every entry). The marker then reads
                        `[3→name]` in that instance's color, and a batch send
                        delivers the entry there. `!` clears the route again.

                                                  *:AskAgentBatchList*
:AskAgentBatchList      List queued questions; choose send now, route to an
                        agent, jump to code, edit or delete.

                                                  *:AskAgentBatchDelete*
:AskAgentBatchDelete    Delete the queued question(s) on the cursor line (asks
                        which one when several sit there).

                                                  *:AskAgentBatchUndo*
:AskAgentBatchUndo      Undo the last delete or clear: put the entries back with
                        their markers. Entries whose buffer was wiped meanwhile
                        are skipped.

                                                  *:AskAgentBatchClear*
:AskAgentBatchClear     Drop every queued question and its marker (undoable).

                                                  *:AskAgentDiff*
:AskAgentDiff [id|head] Review what changed in the newest turn, in turn `id`, or
                        with `head` the whole worktree since HEAD. The interval is
                        two git trees captured at the turn boundaries; see
                        |ask-agent-review|.

                                                  *:AskAgentSnapshot*
:AskAgentSnapshot [id]  Start a manual review interval (no argument); close it
                        with `last` or the id printed on start.

                                                  *:AskAgentHistory*
:AskAgentHistory [query]
                        Search the questions sent before (text, anchor path or
                        label); with no argument lists everything, newest
                        first. Choosing one offers: put in the ask window (the
                        text is prefilled and a fresh anchor is taken), copy
                        text, jump to code, delete.

                                                  *:AskAgentPrompt*
:AskAgentPrompt[!] [name]
                        Ask with a prompt template, rendered against the anchor
                        captured now. Without [name], a picker offers use, copy,
                        edit and delete; see |ask-agent-history|. `!` includes
                        the selection in the anchor.

                                                  *:AskAgentPromptSave*
:AskAgentPromptSave[!] <name>
                        Save the visual selection (or the current line) as a
                        prompt after editing it in an input. `!` writes into
                        `<root>/.ask-agent/prompts.json` so the repository can
                        carry it; otherwise the global file is used.

4. KEYMAPS                                                 *ask-agent-keymaps*

The plugin installs no keymaps: commands are the public interface, mappings belong
to your config. Example:

>lua
vim.keymap.set({ "n", "v" }, "<leader>aq", function()
  require("ask_agent").ask({ visual = vim.fn.mode() ~= "n" })
end, { desc = "Ask agent" })
vim.keymap.set({ "n", "v" }, "<leader>ab", function()
  require("ask_agent").batch_ask({ visual = vim.fn.mode() ~= "n" })
end, { desc = "Ask agent: queue" })
vim.keymap.set("n", "<leader>aB", "<cmd>AskAgentBatchSend<cr>", { desc = "Ask agent: send queue" })
vim.keymap.set("n", "<leader>ai", "<cmd>AskAgentRename<cr>", { desc = "Ask agent: rename" })
<

In the ask window: `<C-s>` sends, `<Esc>`/`<C-c>` cancels, `<CR>` inserts a newline.
In the map: a letter picks that pane, `<Tab>` switches to the list.
In the list: a number (`<M-a>`… past 9) picks where the implementation offers
those; `<Tab>` returns to the map unless the implementation already maps `<Tab>`.

5. PICKERS                                                 *ask-agent-pickers*

`map` draws the split layout of your terminal multiplexer and puts a bold letter
in every pane that hosts an agent. It only covers the window nvim lives in: the
map's value is that a cell's position mirrors the real pane position, which is
lost once other windows (tabs) are mixed in. Agents in other windows are picked
from the list (<Tab>). Zoom does not collapse the map: the full split tree is
used, so an agent in a zoomed-away pane still gets a cell.
A pane covered by a floating window still gets its letter — placed on a visible
cell of that same pane, so nothing is painted over the floating window. Blocks
are laid out compactly (no gaps) and the map is centered on screen.

Only the selected cell (the one under the cursor) is asked for a name: it shows the
agent's name instead of the letter, shortened with "…" when the visible part of the pane
is narrow, and kept as a letter when fewer than `name_min` columns fit. Other cells stay
letters, so long names never crowd the map.

The canvas is a share of the editor clamped to a range of cells. Defaults:
  width = 0.25, min_width = 12, max_width = 32,
  height = 0.20, min_height = 3, max_height = 9,
  name = true, name_min = 3, name_over_badge = true
All of them are set through `map` in setup:
>lua
  require("ask_agent").setup({
    map = { width = 0.3, max_width = 40, name_min = 5 },
  })
<
`name = false` keeps the map to letters only; `name_over_badge = false` always keeps the
state badge next to a name (which then gets cut shorter).

`list` is a plain list of every instance, including ones in other tabs and
windows (the map never shows those), shown through `vim.ui.select` — so it is whatever your config provides
(dropbar, telescope, fzf-lua, snacks, …), with no hard dependency. Each entry
shows the instance glyph in its own stable color, the name, and the cwd. Float
menus are centered rather than attached to the cursor.

Every entry starts with the agent's state, re-probed when the picker opens:

  ●  idle — the agent is free, the question is handled right away

The map draws the agent glyph, the pane letter, then the same status as a colored
badge; a working agent's badge is the rotating spinner. The list colors its leading
status glyph the same way (working/blocked/done/error/gone/idle).
  ◐  working (rotating spinner when the picker animates) — a turn is running; the
     question is still delivered now
     (pi queues it as a follow-up and runs it after the current one), and a
     notice tells you so
  ⚠  blocked — the agent is waiting for you (permission, a question, auth); a
     send would sit in its input unread
  ✓  done — the last turn finished and nobody looked at it yet
  ✗  error — the turn ended in an error
  ○  not running — process exited, socket missing, pane gone/dead, or the pane
     fell back to a shell

Busy rows animate only if the picker implementation supports it. ask-agent passes
`tick` (default 90 ms, see the `tick` setup option) in the `vim.ui.select` opts and
reads `format_item(item, frame)`; that is the contract `ui-menu.nvim` honours
(re-render every tick while waiting for a key).
Other implementations (telescope, fzf-lua, snacks, dropbar) ignore the key and show
the first spinner frame as a static icon. The frames themselves come from the
`spinner` setup option (default: a 10-frame braille spinner); point it at whatever
spinner your picker uses -- e.g. `require("ui_menu").frames` -- so busy rows look
the same in the list and in the map. The map is callback-driven instead of
blocking, so it animates with a uv timer that repaints busy cells and is stopped
when the map closes (no option to set).

The probe may add a short detail next to the dot (the pane's command, "busy",
"process exited", ...). `pi` reports busy via the `busy` field its extension
rewrites on agent_start/agent_settled; other backends may not report it.

Exited instances reported by the backend (`stale_targets()`) are listed at the
end, marked `○ … (exited)`, so a vanished agent is visible instead of absent.
They can never be chosen: picking one only warns and nothing is sent. Disable
with `show_stale = false`. The map shows running instances only.

The list prefers ui-menu.nvim when it is installed (`list_impl = "auto"`, the
default) because that is what provides screen-centering, spinner animation and
<Tab> back to the map. It is recommended, not required: without ui-menu the list
falls back to |vim.ui.select|, and `list_impl = "vim_ui"` forces that fallback.

The map is the primary picker even for a single locatable candidate. Without a
layout adapter, or when no candidate can be located on screen, it falls back to
the list (where <Tab> returns to the map).

6. BACKENDS                                               *ask-agent-backends*

A backend is a table (see `lua/ask_agent/backends/init.lua`):

  id        unique id, e.g. "pi"
  label     shown in prompts
  targets() live instances
  send(target, text) -> ok, err?
  icon      optional glyph
  available() optional: ok, reason?
  status()    optional: ok, reason?
  probe(target) optional: { running, busy?, detail? }  live check for the picker
  stale_targets() optional: table[]             recently exited instances,
                                                shown in the list as "○ not running"

A target is one instance:

  id        stable within its lifetime (pid, socket path, pane id …)
  label     human label, e.g. "pid 12345"
  cwd       project directory (used for matching)
  ts        start time (ordering, alias staleness, color)
  sock      optional delivery endpoint
  rex       optional { session, block }: where it lives, as the rex adapter reads it
  layout    optional: same, in the shape another layout adapter expects
  icon      optional per-instance glyph
  avoid     optional warning to show before sending

Porting to another agent, in five lines: if it runs in a tmux pane, the bundled
factory is all you need (no new file, no plugin change):

  local generic = require("ask_agent.backends.generic")
  require("ask_agent").setup({ extra = {
    generic.tmux({ id = "claude", label = "Claude Code",
                   panes = generic.tmux_panes_by_cmd("claude") }),
  } })

`generic.tmux(opts)` implements `targets()` from the pane list, `send()` (tmux
load-buffer -> paste-buffer -> keys_after, so multi-line text does not turn into
several Enters and quoting is not involved) and `probe()` (pane alive, not dead,
still running the agent). Options: `panes`, `keys_after` (default "Enter"),
`icon`, `busy`. The picker, status dots, batch queue, aliases and the map all come
from the core. Other options for `panes`:
  generic.tmux_panes(match)      all panes as { id, command, cwd, title }
  generic.tmux_panes_by_cmd(pat) filter by pane_current_command

Busy is the one thing tmux cannot report. The recommended route is the agent's own
hooks writing a marker file, read by `generic.state_file(dir, opts)`:

  busy = generic.state_file("~/.cache/ask-agent/claude", { field = "cwd" })

The marker is `<dir>/<slug(key)>.json` holding `{"busy": true}`;
`generic.slug()` is the same escaping a shell hook can reproduce. Missing or
invalid markers mean "unknown" (never blocks sending), and a marker older than
`opts.ttl` (default 3600s) also means unknown, so a crashed session does not look
busy forever. Screen scraping (capture-pane) is possible by writing your own
`busy` function, but it breaks with UI changes, so it is not built in.

Claude Code is bundled: `backends/claude.lua` builds the tmux backend above with
`id = "claude"`, `label = "Claude Code"` and the hook marker directory
`$XDG_CACHE_HOME/ask-agent/claude`. Register it with

  require("ask_agent").setup({ extra = { require("ask_agent.backends.claude").new() } })

and switch with `:AskAgentBackend claude`. `new()` accepts
`id, label, command, command_pattern, state_dir, field, ttl, keys_after`.

Herdr is bundled too: `backends/herdr.lua` lists the agents herdr recognizes in
its panes and reuses herdr's own lifecycle state (idle, working, blocked, done),
so "waiting for you" needs no scraping. It is not registered by default (an agent
under herdr is often visible through another backend as well):

  require("ask_agent").setup({ extra = { require("ask_agent.backends.herdr") } })

One `herdr agent list` per polling pass serves every target; sending goes through
`herdr agent prompt`, viewing through `herdr agent attach`, and `J` through
`herdr agent focus`. A target is `{ herdr = { agent = "<name>" } }`, or a pane id.
Limit what is listed with `M.filter(agent)` before registering the module.
The matching hook script is `extensions/claude-ask-agent-state.sh`; install it as
`~/.claude/ask-agent-state.sh` and add these hooks to `~/.claude/settings.json`:

  UserPromptSubmit -> ask-agent-state.sh busy
  Stop             -> ask-agent-state.sh idle
  SessionStart     -> ask-agent-state.sh idle
  SessionEnd       -> ask-agent-state.sh idle

For non-tmux transports (unix socket, HTTP, file drop, ACP) write a module
returning the table above and register it with `setup({ extra = { mod } })` or
`require("ask_agent.backends").register(mod)`. `tests/backends.lua` in the repo is
a reference: run it from the repo root with
`nvim --headless --clean -l tests/backends.lua` and read how little a
hand-written backend needs.

Individual instances are told apart by a stable color (hash of `id:ts`) and by
an optional name; see |ask-agent-aliases|.

7. INSTANCE NAMES                                   *ask-agent-aliases*

Without a name the directory basename is shown instead of `pid 123`, then the
agent's title or non-pid label. Names live in a JSON file (default under
`stdpath("cache")`). Each backend + real directory keeps slots: a name whose
owner has exited is inherited by the next instance in that directory, while a
name whose owner still runs stays with it — renaming one agent does not rename
the others in the same directory. Only a slot bound to the same pid + start time
is reused. Duplicate names are numbered (`name ·1`, `name ·2`) so the list stays
readable; the numbering covers the whole panel, so two unnamed agents in one
project (both falling back to the directory name) and an alias that collides with
somebody else's default are told apart too. A new instance in a crowded
directory starts unnamed. Set or clear them with |:AskAgentRename|, or press `n`
in the dashboard.

8. LAYOUT ADAPTERS                                   *ask-agent-layouts*

The map needs to know where each instance sits, which is multiplexer-specific:

  id            unique id, e.g. "zellij"
  available()   -> ok, reason?
  window()      -> win, data          "the window nvim itself lives in"
  panes(win)    -> { { id, rect = { x, y, w, h }, kind = "tiled"|"float" }, ... }
                  rect is normalised 0..1, y from the top, later entries paint on top
  match(target) -> pane id|nil        which pane hosts this instance
  cursor()      optional -> pane id   where nvim is focused
  labels(data)  optional -> pane id -> label

Register with `ask.register_layout(mod)`, choose a default with
`setup({ layout = "zellij" })`, or read/set it with `ask.layout()`.

Shipped: `rex` (auto-registered, `lua/ask_agent/layouts/rex.lua`) and a tmux
factory — `require("ask_agent").register_layout(require("ask_agent.layouts.generic").tmux())`.

9. RPC BRIDGE                                         *ask-agent-bridge*

`setup()` starts a Neovim RPC socket and writes a registration file so an agent
can pull live context (unsaved buffers, diagnostics, quickfix) back out:

>sh
nvim --server <sock> --remote-expr \
  'v:lua.require("ask_agent.bridge").handle("<base64(json)>")'
<

Actions: `state`, `buffer`, `diagnostics`, `quickfix`, `list`, `project`,
`agents` (optional `state` filter), `ask` (`{text, backend?, target?}`),
`task` (`{key}`) and `status`. The last four expose
the agent control plane over the same socket, so a terminal agent can list what
is blocked and read the outcome.
Handlers return JSON and never throw. Register extra actions from your config:

>lua
require("ask_agent.bridge").extend("project", function(req)
  return { ok = true, text = my_summary(req.root) }
end)
<

`project` has no built-in implementation: it depends on the host config's
project tooling.

MCP: `bin/ask-agent-mcp` speaks MCP over stdio and forwards the same actions, so
clients that only understand MCP get `agents`/`ask`/`task`/`status` as
tools. Point the client at the script; it finds the nvim instance
working in the current directory (override with `$ASK_AGENT_SOCK`).

10. AGENT STATUS                                            *ask-agent-status*

`ask_agent.state` is the single truth about every instance: the picker, the
dashboard, the statusline and other plugins all read from it, so they cannot
disagree. It polls every `state_interval` ms (default 2000, `state_notify =
false` silences blocked/done/error/exit notifications).

Sources, strongest first:

  rex program status  rex reports each block's OSC 7501 status (`working`,
                      `blocked` with a kind, `done`, `error`, plus `msg` and
                      `progress`). Used for targets carrying
                      `rex = { session, block }`, and streamed through
                      `rex events -s SESSION --json` so changes arrive in
                      milliseconds instead of at the next poll.
  backend probe       `busy` means working, a hook marker may also carry
                      `{state, kind, msg, progress}`; no probe means unknown;
                      `running = false` means gone.
  our send log        a turn that ends after we sent something becomes `done`
                      with `unread = true` (that survives a source that only
                      knows idle, until you acknowledge it).

`rec.source` says which one produced the state: `rex`, `probe`, or `heuristic`
(the last one only knows idle, so the sidebar labels the turn `inferred`).

Notifications                                 *ask-agent-notify-config*

Four state changes reach the user: `done`, `blocked` (with its kinds
`permission` / `question` / `auth`), `error` and `gone` (the agent disappeared
while it still owed you something). `working`/`idle`/`unknown` stay silent.

Each state's copy is a template with `{backend}`, `{name}`, `{msg}` (already
prefixed with ": "), `{state}`, `{kind}` and `{was}` (what a vanished agent was
doing; translate it with `was = { blocked = "在等你", working = "在干活" }` in the
same table); `level` is a key of
`vim.log.levels` as a string, and `sound = true` rings the terminal bell for that
message (unless `notify_belloff = true` and `belloff` contains "all"). Set a state to `false` to silence it,
or give just a string to replace only its text:

>lua
require("ask_agent").setup({
  notify = {
    done = "✅ {name} 完成{msg}",
    error = false,
    blocked = { text = "⏸ {name} 等你{msg}", sound = false },
  },
  notify_gap = 10000,
})
<

The same table can be changed at runtime with
`require("ask_agent.state").configure({ messages = {...}, gap = n })`.
`notify_gap` (default 30000) throttles `done`/`error`/`gone` per agent; a
transition into `blocked` always notifies. `notify_sound = true` makes a sound
on every notification instead of per state.

The plugin writes its own bell, so Neovim's default `belloff=all` does not
silence it (`notify_belloff = true` asks for belloff to win). Terminals are
silent for a bell more often than not, so `notify_sound_cmd` plays a sound
through `vim.system` instead; a plain string runs through `sh -c`. For the
message to be seen, `notify_popup = true` draws a card in the top-right corner
that stays for `notify_popup_ms` instead of one cmdline line the next keystroke
erases (it still lands in `:messages`). A turn typed straight into the agent's
terminal is not announced: only turns this plugin dispatched (or sources that
report `done` themselves, like rex) are detected.

States: `idle`, `working`, `blocked` (kind: permission | question | auth),
`done`, `error`, `unknown`, `gone`. Every transition into `blocked` is reported;
`done` and `error` are throttled to one per 30 s per instance, and `done` is
silent while its local terminal is the current window. They stay in the dashboard
and count in the statusline until acknowledged. An agent that exits while working
or blocked is reported too. Nothing is accepted or rejected automatically: the
agent CLI stays the place where work is approved.

>lua
local ask = require("ask_agent")
ask.agents({ state = "blocked" })          -- what needs me now, most urgent first
ask.on("done", function(rec)
  local c = rec.dispatch and rec.dispatch.correlation
  if c and c.plugin == "todo2" then require("todo2").complete(c.key) end
end)
ask.wait("pi:91841", "done", 600000, function(rec) ... end)
ask.statusline(opts)                         -- "⠹2 ⚠1 ?1 ✓3"; empty when no agent
                                             -- is up. { animated = true, idle = false }
<

*ask-agent-statusline*

The statusline chunk is plain text, most urgent first, no highlight groups:

    ⠹2 ⚠1 ?1 ✓3 ✗1 ●2
    │  │  │  │  │  └ idle agents; only when nothing above is happening
    │  │  │  │  └ failed
    │  │  │  └ finished turns you have not acknowledged yet
    │  │  └ waiting for an answer to a question
    │  └ waiting for your confirmation or credentials
    └ working; the spinner animates at `tick` ms and stops when nothing works

`⚠` and `?` are separate because they ask different things of you: a yes/no versus
an answer. A source reporting `blocked` without a `kind` counts as `⚠`. `✓` goes
away on `a` (acknowledge) or `d` (dismiss). `●` is the fallback and only appears
when the other segments are empty; `statusline({ idle = false })` hides it,
`idle = true` keeps it beside the others, `animated = false` shows a static `◐`.

>lua
  require("ask_agent").statusline()             -- "⠹2 ⚠1 ?1"
  require("ask_agent").statusline({ animated = false })
<

The spinner needs a redraw. In an expression statusline (`%{%...%}`) the plugin's
timer is enough; if you compose the string once, also handle `User
AskAgentStatuslineTick` (fires ~8 times a second while something works, and stops
when nothing does):

>lua
  vim.api.nvim_create_autocmd("User", {
    pattern = "AskAgentStatuslineTick",
    callback = function() vim.o.statusline = build() end,
  })
<

A blocked instance is only reported: the sidebar and statusline show it, and
`<CR>` jumps to its terminal. It is answered in that terminal, not from Neovim.
Tasks and unread marks are journaled to
`<cache>/ask-agent/journal.json` (1 s debounce, week-old tasks dropped, disable
with `persist = false`), so a restart keeps pending work visible.

`rec.dispatch = { ts, correlation, task_id }` records when we last sent, the
opaque correlation table given to `ask.send(text, { correlation = { plugin, key } })`,
and the id the agent was sent with; it comes back with every event for that
instance. Every send is also appended to `rec.tasks` (newest first, the last 20):
`{ text, ts, status, correlation, anchor, task_id, turn_id, has_answer }`
with `status` = `running` / `done` / `error`. `task_id` lets an answer find its
question (`ask.answer()`), `turn_id` is the snapshot turn the send opened (what
`v` diffs — no more "no turn recorded" for a plain ask), `has_answer` is set when
the agent handed the answer back.

11. DASHBOARD                                        *ask-agent-dashboard*

|:AskAgentDashboard| opens a sidebar tree on the right (40 columns; pass
`{ pos = "left", width = 30 }` to `ask.dashboard()` to move or resize it). Top
level: agent instances, most urgent first. Children: that agent's tasks — questions still queued for it (`◐`) and
turns already sent (spinner running, `✓` done, `✗` failed). Agent glyphs: `●`
idle, spinner working, `⚠` blocked, `✓` done, `✗` error, `◌` unknown, `○` exited.
Tasks are connected to their agent with `│ ├ └` lines (`tree_lines`, default
true); the rail below the fold marker keeps that connection continuous.
The path from an agent to the task it is running right now is green: `running_line`
defaults to `"dash"`, a dashed line whose bright cells march down one step per
spinner frame; `"pulse"` sends a single bright cell with a trailing mid shade,
`"static"` draws a dim→bright gradient and `"off"` leaves plain tree lines. Both
degrade gracefully: below 20 columns the connectors give way to indentation, and
without 'termguicolors' the path is one flat green. The connectors use the
`AskAgentTree` highlight group (linked to `Comment`) and the path the
`AskAgentRunDim` / `AskAgentRunMid` / `AskAgentRunBright` groups.

`<CR>` on an agent opens it in a viewer float (see |ask-agent-viewer|). On a task it
follows the source of that task: work another tool dispatched (todo2, a plugin over
the bridge) jumps to the code anchor it was filed at, because that is where the task
lives; a turn of your own opens the agent instead, since it lives in the agent's
transcript and the anchor is only where your cursor happened to be. A dispatched
task jumps to the anchor it was filed at (the row shows it as `file.lua:42`; an
anchor whose buffer was wiped or whose buffer number now belongs to another file
is reopened from its path). An anchor that lies outside the current project — the
file is not under this tab's working directory, or under the same repository root
as it — opens in a tab of its own, that tab taking the anchor's working directory,
so jumping to another project never replaces what you were looking at. An anchor taken
inside a diff is treated as what it is: a viewer's plugin buffer (a `codediff://…` or
this plugin's own `ask-agent-diff://…` name) has no path, so the anchor carries only
its name and jumping to it says why it cannot; a file inside a tree extracted for `v`
stays the path (the agent is told it is a copy of the project at that revision and
should read it, not edit it) and jumping resolves its relative path back to the
project's own file. `J` goes the
other way — out to the terminal window the agent really runs in, through a jump
adapter (see |ask-agent-jump|). `s` sends, `S` flushes every queued question
routed to that agent, `a` acknowledges, `d` deletes a
queued task or dismisses a dead agent, `v` reviews (the turn behind that task, or
the whole worktree for an agent), `o` shows the question and the answer the agent
handed back (|ask-agent-results|), `y` copies the task text, `f` toggles "only
agents that need you", `<Tab>` (or `za`) folds its tasks, `R` renames, `?` opens
the key help float, `r` polls now, `q` closes.

Every key is configurable, and `false` unbinds it:

  >lua
  require("ask_agent").setup({
    keys = {
      rename = "R",             -- open, jump, send, send_queue, ack, dismiss,
      answer = false,           -- rename, diff, answer, copy, filter, fold,
      fold = { "<Tab>", "za" }, -- refresh, close, help
    },
  })
  <
The help float shows the keys you actually bound, in the same order, so it never
lies about the buffer. Defaults follow vim conventions: `n`/`N` stay free for
next match, `r` refreshes, `y` yanks, `d` deletes/dismisses, `v` views a diff,
`R` renames, `J` jumps out to the agent's own window.

The winbar counts what wants you (`⚠2 ✓1 ◐3`) and lists the sort order in the
`?` help: blocked/error, done, working, idle. A freshly opened sidebar grows once
(up to 48 columns) when a row would be truncated; below 44 columns the backend
column is dropped before the message is cut. Finished tasks show how long the turn
took; running ones show how long it has been going. A long list keeps the newest
`task_limit` rows and folds the rest into `… N older tasks  ⏎ show` (`<CR>` or
`<Tab>` on that row expands it; folding the agent resets the slice). Finished
tasks older than `task_ttl` are pruned from the journal on save/load, except the
newest task of each agent. A turn dispatched by another tool (todo2, a plugin)
shows that tool's key dim in brackets on the row — `[t42]` — so several sources
can share one agent and still be told apart. The view is event-driven (state + queue notifications) and keeps the
cursor where it was; the spinner is the only timer and only runs while a turn is
in progress.

                                                          *ask-agent-jump*
Jumping out to the agent: a float is a second display, `J` brings the original
one to the front. A jump adapter is the whole interface:

>lua
  require("ask_agent").register_jump({
    id = "kitty",
    priority = 10,                 -- higher wins when several adapters match
    can = function(target) return target.kitty ~= nil end,
    focus = function(target)
      vim.system({ "kitty", "@", "focus-window", "--match", target.kitty })
    end,
  })
<

rex (`target.rex`), herdr (`target.herdr`, via `herdr agent focus`) and tmux
(`target.pane`) ship built in. The first matching
adapter (highest `priority`) handles the jump. The rex adapter runs three steps,
because a Rex app shows one session and one window (tab) at a time: `rex block
inspect` finds the block's `window_id`, `rex window focus <window> -s <session>`
switches tabs, `rex block focus <block> -s <session>` puts the cursor in the pane,
and finally `rex do session.select session_id=...` (the app-side action;
`REX_CLIENT` picks the app when set) makes the session visible. Failures become
warnings. `ask.jump()` does the same from Lua, `ask.jumps()` lists the adapters.

                                                          *ask-agent-viewer*
Which terminal or multiplexer shows an agent is user-specific, so the sidebar
knows a provider contract: a "viewer" is the host's own second-display command.

>lua
ask.register_view({
  id = "herdr",
  priority = 10,                          -- higher wins when several match
  can = function(target) return target.herdr ~= nil end,
  cmd = function(target, opts)
    return { cmd = { "herdr", "agent", "attach", target.herdr.id } }
  end,
})
<

`cmd` returns `{ cmd, env, width, height, note }`; everything but `cmd` is
optional (`env` unsets inherited variables, `note` is shown on the float). The
first matching provider (highest `priority`) handles the target. rex
(`target.rex = { session, block }`), tmux (`target.pane`) and herdr
(`target.herdr = { agent }`, or `{ pane }`) ship built in, and `:checkhealth
ask_agent` lists them.

The float opens in terminal mode, like |:terminal|: open it and type. The float
adds only these keys, and its footer always shows them:

  `<C-q>`    normal and terminal mode: close the float, agent keeps running
  `q`        normal mode: same as `<C-q>`
  `i`        normal mode: start typing into the agent (writable viewers only)

`<Esc>` is left to the agent on purpose (there it means "interrupt"); use
`<C-q>` to leave, or `<C-\><C-n>` then `q` to copy text first. The title shows
`agent · provider · state` (dropping parts when narrow) and the footer shows the
keys plus `read only`, the provider's `note`, and `clipped WxH→WxH` when the
agent's grid does not fit the screen.

`view.readonly = true` makes a watch-only float: tmux gets `-r`, the rex mirror
drops stdin, herdr keeps its non-takeover display.

One agent has one float: opening it again focuses the open float. The float is
tied to the agent's lifecycle: when the state layer reports the agent gone, or
the host client exits, the float closes itself. The rex mirror also notices a
block whose shell outlives the agent (`Ctrl-C` in the float) and closes.

The float is clipped, never wrapped, when the agent's terminal is larger than
the screen: the top of a taller grid is cropped and long lines are cut at the
right, and the footer says so. A float never takes more than 85% of the editor's
width and 75% of its height even when the agent's grid is bigger (`view.width` /
`view.height`; 1 fills the screen).

tmux is attached with `-f ignore-size` and the float is sized to the session's
grid (window rows plus the status line), so viewing never reflows the panes
somebody else is looking at. A viewer is a second client, not a mirror: a
full-screen TUI may still repaint when the screen is too small for the session,
and `destroy-unattached` can end a tmux session when its last client leaves.

Rex has no attach, so that viewer mirrors instead: `bin/rex-view.py` asks rex for
the block's screen (visible rows only, not scrollback) as VT text, repaints whole
frames and forwards typed keys with `com.superlogical.terminal.write`. Whole
frames keep the picture correct even when the float and the block disagree about
size. Repaints are driven by the block's data connection
(`rex.data.terminal.v1`): new output paints immediately (at most every 20 ms), an
idle block costs nothing and a keypress echoes in about 25 ms. A once-a-second
repaint remains as a safety net. The helper polls the block's process once a
second and exits (closing the float) when the block closes, when its program
exits, or when the agent exited but the block's shell survived. tmux and herdr
need only their own binary; the rex mirror runs `bin/rex-view.py` with `python3`
(override with `vim.g.ask_agent_python`).

The rex CLI itself is found even when it is not on `$PATH` (Neovim started from
a herdr pane or a launcher that does not source Rex's shell setup): the plugin
uses `vim.g.ask_agent_rex`, then `$ASK_AGENT_REX`, then `$PATH`, then the newest
installation under `~/Library/Application Support/com.superlogical.rex/RexServer/
Versions/*/Rex Server.app/Contents/MacOS/rex`. ``:checkhealth ask_agent``
reports which one it picked.

12. BATCH QUESTIONS                                         *ask-agent-batch*

Queue questions while browsing instead of sending them immediately. Each queued
entry keeps the anchor captured at the time (cwd, file, line, selection), is
drawn in the source buffer as a `»` sign plus end-of-line virtual text, and
follows buffer edits through its extmark. |:AskAgentBatchSend| sends every entry
as its own message; |:AskAgentBatchSend| `3,5-7` sends only those.

|:AskAgentBatchRoute| routes entries to a specific agent instance of the current
backend (|:AskAgentBackend| switches backends); the marker shows `[3→name]` in that
instance's color, so one queue can be split across several running agents: a send
groups by route, routed entries go to their own agent and the unrouted rest shares
one picker.

In the ask window: `<C-s>` queues the question, `<Esc>`/`<C-c>` cancels.
Use |:AskAgentBatchList| to jump to an entry's code, edit its text or delete it,
or |:AskAgentBatchDelete| to drop the entry under the cursor. Every delete and
|:AskAgentBatchClear| can be reversed with |:AskAgentBatchUndo| (the last 20
operations are kept in memory, as is the queue itself — the current session
only).

12. CHANGE REVIEW                                         *ask-agent-review*

|:AskAgentDiff| answers "what did the agent change?" with evidence instead of a
heuristic. At a turn boundary the plugin records the worktree as a git tree: it
copies `.git/index` to a temp file, runs `git add -A` against that copy and
`git write-tree`. The real index and refs are never touched, nothing is staged
or committed. When the turn ends the second tree is recorded, and the review is
`git diff <start> <finish>` — an immutable interval anyone can re-run.

What can be bracketed:

  whole worktree vs HEAD   |:AskAgentDiff| `head`, no turn needed
  one sent question        tree before the send, tree when the turn ends
  typed-in turns           the state layer's working -> done (trigger_state)
  one tool call            needs agent cooperation; out of scope

Limits worth knowing: edits you make yourself inside the window are
indistinguishable from the agent's; `blocked` does not close a turn; a second
question sent while a turn is open widens the same window instead of starting a
new one.

Viewers are adapters, picked in order: `diffbandit` (installed), `nvim.difftool`
(Neovim 0.12+, zero install), `codediff` (installed; trees wrapped in ref-less
commits), `builtin` (always: one `filetype=diff` buffer plus a quickfix list of
changed files). Pin one with `review.backend`. With `include_unsaved` (default)
modified buffers are written into a throwaway copy of the end tree, so the
review shows what is in the editor; the builtin backend diffs trees and
therefore shows written bytes only.

Snapshots live in `stdpath("cache")/ask-agent/snapshots.json` (newest 50, 7 days
by default). Use |:AskAgentSnapshot| to bracket work that never passed through
the plugin, `:AskAgentDiff <id>` to review a stored turn, and `ask.turns()` to
list them.

13. HISTORY AND PROMPTS                                  *ask-agent-history*

Both features assemble the same payload the agent always receives — text plus
anchor — so neither one reads the agent back or parses its transcript.

History is written on the send path, after the backend accepted the text, so a
failed send is never recorded as delivered. Each record keeps the text, the
anchor (`path:line`, selection, cwd), the backend, the instance label and the
caller's `correlation` (shown as `todo2 t42` when the question came from a task). The
records survive restarts; jumping to the code falls back to opening the file by
path when the original buffer is gone.

  >lua
  ask.history({ query = "rename", limit = 10 })  -- newest first
  ask.answer(task_id)          -- final answer to that task, or nil
  ask.show_answer(task)        -- question + answer in a float (`o`)
  <
Filter by `cwd`, `backend` or free `query` (text, path, label).

                                                       *ask-agent-results*
Answers are opt-in (`results = true`, off by default): they turn the ledger into
a little transcript, which history alone does not promise. The agent is sent
`{"text": ..., "task_id": ...}`, and it hands the answer back keyed by that id —
the answer is never parsed out of a transcript:

  pi       the nvim-context extension quotes the id back over the nvim bridge
           (`action=result`) when a run ends; the last assistant text of that run
           becomes the answer. A run only claims tasks sent before it started, so
           a queued follow-up cannot receive the previous run's answer.
  hooks    write `{"task_id": "...", "text": "..."}` (or, without an id,
           `{"cwd": "/repo", "question": "...", "text": "..."}`) into
           `<cache>/ask-agent/results/` (the older `$XDG_CACHE_HOME/ask-agent/
           results/` is still read); imported on dashboard open, on `o`, and
           when an agent reports done. Without an id the answer goes to the task
           whose question appears in `question`, else to the newest task of the
           agent in that `cwd`. Undeliverable files are kept, not deleted; a week
           later they are dropped, so a Stop hook cannot fill the directory
           forever.
           `extensions/claude-ask-agent-state.sh done` does this from a Claude
           Code Stop hook (it also quotes the transcript's last user message as
           `question`, so two agents in one directory keep their own answers).

Answers are capped like questions (`history.max_text`) and pruned with the record
(`history.keep`). With `results` off, `o` says there is no answer and nothing is
stored.

Prompts are JSON files, `{ version = 1, prompts = [{ name, text, desc }] }`. A
project prompt overrides a global one with the same name:

  global    `stdpath("cache")/ask-agent/prompts.json`
  project   `<root>/.ask-agent/prompts.json`  (root = git root, else nearest
            ancestor containing `.ask-agent`)

Placeholders are replaced against the anchor of that ask; unknown ones are left
verbatim so a typo stays visible:

  {{file}}         anchor path, e.g. `src/a.lua`
  {{line}}         anchor line;  {{col}} column
  {{filetype}}     buffer filetype
  {{cwd}}          anchor cwd
  {{selection}}    visual lines joined with newlines (empty without selection)
  {{diagnostics}}  number of diagnostics on the anchor buffer
  {{modified}}     `unsaved` when the buffer had unsaved changes

  >lua
  ask.prompt("review")                        -- render + open the ask window
  ask.prompt_save("review", text, { project = true })
  ask.prompt_remove("review")
  ask.prompts(cwd)                            -- visible templates
  <
A history record can be re-asked through `ask.ask({ default = rec.text })`; it is
the same code path as the ask window, with a fresh anchor.

14. WHAT A QUESTION CARRIES                                *ask-agent-anchor*

The anchor is captured `before` any UI opens — a float or a picker would change
the window — so it is what you were looking at when you pressed the key. It is
plain data, agent-agnostic: every backend is handed the same fields.

  cwd, tab        working directory of this tab (tab_workspace gives each tab its
                  own via :tcd) and its number
  root, rel,      repository root, the file relative to it (what `git grep`, an
  branch          editor or a compiler expects), and the branch. `path` stays
                  relative to `cwd`; `rel` wins when both exist
  path, line, col the file (~ folded) and the cursor position
  filetype        language for the code fence
  modified        the buffer has unsaved changes: the file on disk is `not` what
                  is on screen
  diagnostics     error/warn diagnostics on the cursor line, at most three
  context         the source around the cursor, but only for a buffer whose text
                  the agent cannot read from disk: unsaved, or with no path (a
                  diff buffer). A saved file is read by the agent itself
  visual          in visual mode: start/end line plus the selected text
  virtual         a plugin buffer (codediff://…, fugitive://…) travels as a name
                  only, never as a plausible-looking path
  snapshot        a copy inside a tree extracted for `v`: readable, but marked
                  "read it, don't edit it"
  digest          the project overview (see below)

Two things ride along: the `digest` says what this project is, the `anchor` says
where in it the question was asked. The digest comes from project-overview.nvim
(structure, size, build entry, conventions — no source); it is fetched when the
question is `sent`, cached per root, and never fails a question: no plugin, a
broken collect or `enabled = false` just means the block is absent.

  >lua
  require("ask_agent").setup({
    project = {
      enabled = true,   -- attach the digest when project-overview.nvim is installed
      wait_ms = 1500,   -- how long a cold first collect may take
      full = false,     -- also deps / conventions / size detail
      ttl = 300,        -- seconds a digest is reused
    },
  })
  <

A text-only backend receives the question and this block as one string:

    why does this loop skip the last item?

    <nvim-context src/parser.rs:42 · branch main · tab 1 · unsaved>
    cwd: /home/me/proj
    root: /home/me/proj
    diagnostics:
      rust-analyzer:42: error: mismatched types
    context (lines 22-62):
    ```rust
    ...
    ```
    </nvim-context>

Inlining is deliberately narrow — copying a saved file only costs tokens and goes
stale, while an unsaved buffer `is` a different file on disk. Selections are
capped with a `(clipped)` marker:

  >lua
  require("ask_agent").setup({
    context_lines = 20,        -- lines around the cursor; 0 disables the snippet
    max_selection = 200,       -- lines of selected text that travel inline
    max_selection_bytes = 8192,
  })
  <

The journal keeps the coordinate, not the source: `context`, `diagnostics`,
`digest` and `visual.lines` are dropped before a task is recorded, so journal.json
and history.json stay small and a jump still lands on the line.
`ask.goto_anchor(a)` reads the same fields back.

The anchor is a *protocol field*, not prose: pi receives `text` (the question
alone) plus `anchor` as JSON, and renders the block itself — the same layout as
message.block in Lua — so the question stays clean in its transcript, in the
history ledger, in the `o` answer float and in a yanked task. A backend without
`anchor_payload = true` still gets `message.render` output (question + block).

15. API                                                    *ask-agent-api*

>lua
local ask = require("ask_agent")
ask.setup(opts)
ask.ask({ visual = true })
ask.batch_ask({ visual = true })
ask.batch_send({ ids = "3,5-7" }) / ask.batch_route(ids, opts)
ask.batch_list() / ask.batch_clear() / ask.batch_count()
ask.send(text, opts)                    -- { backend, target, correlation, anchor,
                                        --   pick, on_sent }
ask.task(key)                           -- newest task handed to that agent
ask.agents({ state = "blocked" })       -- instances, most urgent first
ask.on(event, fn)                       -- state | update | done | blocked | error
ask.wait(key, want, ms, cb)             -- one-shot wait; cb(nil) on timeout
ask.statusline() / ask.dashboard(opts)  -- sidebar; { pos = "right", width = 40 }
ask.goto_anchor(a)                      -- jump to a captured anchor
                                        -- (pick = true in send: ask which instance)
ask.diff({ id = "..." })                -- review a turn / { head = true }
ask.snapshot(id)                        -- start (nil) / close ("last") an interval
ask.turns(cwd)                          -- recorded turns, newest first
ask.history({ query = "..." })          -- delivered questions, newest first
ask.answer(task_id)                     -- final answer (results = true), or nil
ask.show_answer(task)                   -- question + answer in a float
ask.prompts(cwd)                        -- prompt templates (global + project)
ask.prompt(name, opts)                  -- ask with a template
ask.prompt_save(name, text, opts) / ask.prompt_remove(name, opts)
ask.use("pi") / ask.current()
ask.list() / ask.targets("pi")
ask.list_impl("auto"|"ui_menu"|"vim_ui")
ask.pin(id) / ask.pinned()
ask.alias(target, name) / ask.alias_of(target)
ask.picker("map"|"list")
